import java.util.StringTokenizer
import kotlin.math.max

private val rowDir = listOf(-1, 1, 0, 0)
private val colDir = listOf(0, 0, -1, 1)
private lateinit var map: Array<BooleanArray>

private fun dfs(row: Int, col: Int): Int {
    map[row][col] = false
    var size = 1
    for (i in 0..3) {
        val newRow = row + rowDir[i]
        val newCol = col + colDir[i]

        if (map[newRow][newCol]) {
            size += dfs(newRow, newCol)
        }
    }
    return size
}


fun main() = with(System.`in`.bufferedReader()) {
    val (row, col) = readLine().split(" ").map { it.toInt() }
    map = Array(row + 10) { BooleanArray(col + 10) }

    repeat(row) { r ->
        val st = StringTokenizer(readLine())
        val line = Array(col) { st.nextToken().toInt() }
        line.forEachIndexed { c, v ->
            map[r + 1][c + 1] = v == 1
        }
    }

    var maxSize = 0
    var count = 0

    for (j in 1..row) {
        for (i in 1..col) {
            if (map[j][i]) {
                val size = dfs(j, i)
                maxSize = max(maxSize, size)
                count++
            }
        }
    }

    println(count)
    println(maxSize)
}
