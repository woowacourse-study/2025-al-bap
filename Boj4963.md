package com.example.algorithm.backjoon.kotlin.dfs

import java.util.StringTokenizer

// https://www.acmicpc.net/problem/1012
/**
 
private lateinit var map: Array<BooleanArray>
private val rowDir = listOf(-1, 1, 0, 0)
private val colDir = listOf(0, 0, -1, 1)

private fun dfs(row: Int, col: Int) {
    map[row][col] = false

    for (i in 0..3) {
        val newRow = row + rowDir[i]
        val newCol = col + colDir[i]

        if (map[newRow][newCol]) {
            dfs(newRow, newCol)
        }
    }
}

fun main() = with(System.`in`.bufferedReader()) {
    val t = readLine().toInt()

    repeat(t) {
        val (col, row, k) = readLine().split(" ").map { it.toInt() }
        map = Array(row + 2) { BooleanArray(col + 2) }

        repeat(k) {
            val st = StringTokenizer(readLine())
            val (x, y) = Array(2) { st.nextToken().toInt() }
            map[y + 1][x + 1] = true
        }

        var answer = 0
        for (r in 1..row) {
            for (c in 1..col) {
                if (map[r][c]) {
                    answer++
                    dfs(r, c)
                }
            }
        }

        println(answer)
    }
}
